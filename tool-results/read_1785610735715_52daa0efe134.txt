  1369→function compareSemver(v1, operator, v2) {
  1370→  let versionDiff = 0;
  1371→  if (v1 === "latest") {
  1372→    versionDiff = 1;
  1373→  } else {
  1374→    if (/^[^\d]/.test(v1)) {
  1375→      v1 = v1.substring(1);
  1376→    }
  1377→    if (/^[^\d]/.test(v2)) {
  1378→      v2 = v2.substring(1);
  1379→    }
  1380→    const [major1, minor1 = 0, patch1 = 0] = v1.split(".").map(Number);
  1381→    const [major2, minor2 = 0, patch2 = 0] = v2.split(".").map(Number);
  1382→    if (Number.isNaN(major1) || Number.isNaN(major2)) {
  1383→      throw new Error("The major version is required.");
  1384→    }
  1385→    if (major1 !== major2) {
  1386→      versionDiff = major1 - major2;
  1387→    } else if (minor1 !== minor2) {
  1388→      versionDiff = minor1 - minor2;
  1389→    } else if (patch1 !== patch2) {
  1390→      versionDiff = patch1 - patch2;
  1391→    }
  1392→  }
  1393→  switch (operator) {
  1394→    case "=":
  1395→      return versionDiff === 0;
  1396→    case ">=":
  1397→      return versionDiff >= 0;
  1398→    case "<=":
  1399→      return versionDiff <= 0;
  1400→    case ">":
  1401→      return versionDiff > 0;
  1402→    case "<":
  1403→      return versionDiff < 0;
  1404→    default:
  1405→      throw new Error(`Unsupported operator: ${operator}`);
  1406→  }
  1407→}
